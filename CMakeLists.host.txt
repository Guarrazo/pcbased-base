cmake_minimum_required(VERSION 3.13)

# CMake alternativo para compilar SOLO los tests en el host de desarrollo
# (Windows/Linux/macOS), SIN necesitar el toolchain de Nintendo Switch.
# 
# Uso:
#   mkdir build-host
#   cd build-host
#   cmake .. -C ../CMakeLists.host.txt
#   cmake --build .
#   ctest
#
# Este build NO genera el ejecutable final para Switch, solo los tests
# que validan los componentes independientes de plataforma (decoder,
# IR builder, emitter, etc.).

project(pc-arcade-switch-tests C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 17)

# Forzar build de tests
set(PAS_BUILD_TESTS ON)

# --- Núcleo independiente de plataforma -------------------------------------
# Solo los módulos que NO dependen de libnx/Switch
add_subdirectory(src/core)
add_subdirectory(src/cpu)
add_subdirectory(src/os)
add_subdirectory(src/patch)
add_subdirectory(src/profiles)

# --- Tests -------------------------------------------------------------------
enable_testing()
add_subdirectory(tests)
